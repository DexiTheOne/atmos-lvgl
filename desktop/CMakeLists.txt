cmake_minimum_required(VERSION 3.20)
project(atmos_desktop C CXX)
set(CMAKE_C_STANDARD 11)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/lvgl/lvgl.h")
  message(FATAL_ERROR "Run build.ps1 to download the pinned LVGL dependency")
endif()
# Build LVGL's C sources directly. LVGL's own CMake (9.6.0) also compiles optional
# C++ glTF sources whose headers are missing from the v9.6.0 tag.
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/lvgl/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/vendor/lvgl" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
add_executable(atmos main_win32.c atmos.c ui_fonts.c)
target_link_libraries(atmos PRIVATE lvgl gdi32 user32 m)
target_compile_options(atmos PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME native_render COMMAND atmos --self-test)
add_test(NAME boot_intro COMMAND atmos --test-intro)
set_tests_properties(native_render PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
