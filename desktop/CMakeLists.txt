cmake_minimum_required(VERSION 3.20)
project(atmos_desktop C CXX)
set(CMAKE_C_STANDARD 11)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "")
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/lvgl/lvgl.h")
  message(FATAL_ERROR "Run build.ps1 to download the pinned LVGL dependency")
endif()
add_subdirectory(vendor/lvgl EXCLUDE_FROM_ALL)
add_executable(atmos main_win32.c atmos.c ui_fonts.c)
target_link_libraries(atmos PRIVATE lvgl gdi32 user32 m)
# Unsuffixed literals are float, so model maths stays single precision (ESP32-S3 FPU).
target_compile_options(atmos PRIVATE -Wall -Wextra -Werror -fsingle-precision-constant)
enable_testing()
add_test(NAME native_render COMMAND atmos --self-test)
set_tests_properties(native_render PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
